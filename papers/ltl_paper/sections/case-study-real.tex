\subsection{Real-Valued Accumulator}
\label{sec:case-real}

\paragraph{Reference system $Z_{\mathbb{R}}$.}
Behavior: discrete-time running sum. Start at $0$; on real input $u$ replace the state by $s+u$; output the current sum. (The clock is discrete; values are real.)

\paragraph{Compiled $\Phi_{\mathrm{dyn}}$.}
For every $s,u\in\mathbb{R}$:
\begin{align*}
  &\text{(readout)} && G\bigl(\mathit{state}(s) \rightarrow \mathit{output}(s)\bigr), \\
  &\text{(guided)} && G\bigl((\mathit{state}(s)\wedge\mathit{input}(u)) \rightarrow X(\mathit{state}(s{+}u))\bigr), \\
  &\text{(autonomous)} && G\bigl((\mathit{state}(s)\wedge\mathit{input}(\mathrm{none})) \rightarrow X(\mathit{state}(s))\bigr).
\end{align*}

\paragraph{Three buildable systems (shown explicitly).}

\textit{Simple --- $Z_{\mathbb{R}}^{\mathrm{S}}$.}
Identical to Table~\ref{tab:z-real}; identity maps.

\paragraph{Classical homomorphism check for $Z_{\mathbb{R}}^{\mathrm{C}}$.}
Take $h_S(s,\ell,t,e)=s$, $h_I=\mathrm{id}$, and $h_O=\mathrm{id}$. Surjectivity: every sum $s$ is realized by $(s,0,0,0)$. Transition preservation: for real input $u$,
\[
  h_S\bigl(\delta_{\mathrm{C}}((s,\ell,t,e),u)\bigr)
  = h_S(s{+}u,\,u,\,t{+}1,\,e{+}u^{2})
  = s{+}u
  = \delta_{\mathbb{R}}(s,u)
  = \delta_{\mathbb{R}}\bigl(h_S(s,\ell,t,e),\,h_I(u)\bigr),
\]
and likewise for $\mathrm{none}$ (state unchanged). Readout: $\rho_{\mathrm{C}}(s,\ell,t,e)=s=\rho_{\mathbb{R}}(h_S(s,\ell,t,e))$. Hence $h$ is a Wymore system homomorphism $Z_{\mathbb{R}}^{\mathrm{C}}\to Z_{\mathbb{R}}$, so $Z_{\mathbb{R}}^{\mathrm{C}}\models_{\mathrm{dyn}}\Phi_{Z_{\mathbb{R}}}$ by the bi-implication. As in the discrete examples, $\Phi_{Z_{\mathbb{R}}}$ does not mention ticks or energy: those channels may change freely provided the projected sum obeys the accumulator table. Exact real arithmetic is used in the formalization; floating-point is not identified with $\mathbb{R}$, and unbounded real FO-LTL checking is not claimed to be decidable.
